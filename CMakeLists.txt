cmake_minimum_required(VERSION 3.24)
project(Google-Photo-Uploader VERSION 0.1.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
enable_testing()
add_executable(core_tests tests/core_tests.cpp)
target_include_directories(core_tests PRIVATE src)
add_test(NAME core COMMAND core_tests)
add_executable(phone_model_tests tests/phone_model_tests.cpp)
target_include_directories(phone_model_tests PRIVATE src)
add_test(NAME phone_model COMMAND phone_model_tests)
if(WIN32)
  set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
  # 창 제목에 표시할 빌드 이름 (CI: main.<run>-<커밋>)
  set(GPU_VERSION "dev" CACHE STRING "Google-Photo-Uploader build name")
  configure_file(src/version.h.in ${CMAKE_CURRENT_BINARY_DIR}/generated/version.h @ONLY)
  add_executable(Google-Photo-Uploader WIN32 src/main.cpp src/organizer.cpp src/adb.cpp src/uploader.cpp)
  target_include_directories(Google-Photo-Uploader PRIVATE src ${CMAKE_CURRENT_BINARY_DIR}/generated)
  target_compile_definitions(Google-Photo-Uploader PRIVATE UNICODE _UNICODE NOMINMAX WIN32_LEAN_AND_MEAN _WIN32_WINNT=0x0A00 WINVER=0x0A00)
  target_link_libraries(Google-Photo-Uploader PRIVATE ole32 shell32 shlwapi comctl32 propsys bcrypt advapi32 user32 gdi32 uuid)
  # CI용 가짜 adb (폰 흉내)
  add_executable(fake_adb tests/fake_adb.cpp)
  target_compile_definitions(fake_adb PRIVATE UNICODE _UNICODE NOMINMAX WIN32_LEAN_AND_MEAN)
  if(MSVC)
    target_compile_options(fake_adb PRIVATE /EHsc /utf-8)
  else()
    target_compile_options(fake_adb PRIVATE -municode)
    target_link_options(fake_adb PRIVATE -municode -static)
  endif()
  if(MSVC)
    target_compile_options(Google-Photo-Uploader PRIVATE /W4 /EHsc /utf-8)
    target_link_options(Google-Photo-Uploader PRIVATE "/MANIFEST:EMBED" "/MANIFESTINPUT:${CMAKE_CURRENT_SOURCE_DIR}/src/app.manifest")
    set_property(TARGET Google-Photo-Uploader PROPERTY MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
  else()
    target_compile_options(Google-Photo-Uploader PRIVATE -Wall -Wextra -Wno-missing-field-initializers -municode)
    target_link_options(Google-Photo-Uploader PRIVATE -municode -static)
  endif()
endif()
